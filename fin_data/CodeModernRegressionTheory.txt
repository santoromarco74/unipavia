    A bit of coding on regression trees
    
    hyper_grid <- expand.grid(
     minsplit = seq(5, 20, 1),
     maxdepth = seq(8, 15, 1)
    )
    
    models <- list()
    for (i in 1:nrow(hyper_grid)) {
         minsplit <- hyper_grid$minsplit[i]
     maxdepth <- hyper_grid$maxdepth[i]
    models[[i]] <- rpart(
     formula = Sale_Price ~.,
     data = ames_train,
     method = "anova",
     control = list(minsplit = minsplit, maxdepth = maxdepth)
     )
    }
    get_cp <- function(x) {
     min <- which.min(x$cptable[, "xerror"])
     cp <- x$cptable[min, "CP"]
    }
    get_min_error <- function(x) {
     min <- which.min(x$cptable[, "xerror"])
     xerror <- x$cptable[min, "xerror"]
    }
    hyper_grid %>%
     mutate(
     cp = purrr::map_dbl(models, get_cp),
     error = purrr::map_dbl(models, get_min_error)
     ) %>%
     arrange(error) %>%
     top_n(-5, wt = error)
    
    optimal_tree <- rpart(
     formula = Sale_Price ~.,
     data = ames_train,
     method = "anova",
     control = list(minsplit = 11, maxdepth = 8, cp = 0.01)
     )
    pred <- predict(optimal_tree, newdata = ames_test)
    RMSE(pred = pred, obs = ames_test$Sale_Price)  //RMSE=40k means price is (312±1.96∗40)
    
    Bagging
    
    set.seed(123)
    bagged_m1 <- bagging(
     formula = Sale_Price ~.,
     data = ames_train,
     coob = TRUE
    )
    bagged_m1
    
    ntree <- 10:50
    rmse <- vector(mode = "numeric", length = length(ntree))
    for (i in seq_along(ntree)) {
     set.seed(123)
          model <- bagging(
     formula = Sale_Price ~.,
     data = ames_train,
     coob = TRUE,
     nbagg = ntree[i]
    )
     rmse[i] <- model$err
    }
    plot(ntree, rmse, type = 'l', lwd = 2)
    abline(v = 25, col = "red", lty = "dashed")
    
    pred <- predict(bagged_m1, ames_test)
